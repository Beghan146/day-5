#include <stdio.h>

int main() {
    int arr[] = {12, 45, 23, 67, 34, 89, 10};
    int n = sizeof(arr) / sizeof(arr[0]);
    int i, j, temp;

       for (i = 0; i < n - 1; i++) {
        for (j = 0; j < n - i - 1; j++) {
            if (arr[j] < arr[j + 1]) {
                // Swap
                temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
    printf("Array in descending order:\n");
    for (i = 0; i < n; i++) {
        printf("%d ", arr[i]);
    }

    return 0;
}
