c program for multiply 3 arrays



#include <stdio.h>

int main() {
    int size = 5;
    int A[] = {1, 2, 3, 4, 5};
    int B[] = {2, 3, 4, 5, 6};
    int C[] = {3, 4, 5, 6, 7};
    int result[size];

       for (int i = 0; i < size; i++) {
        result[i] = A[i] * B[i] * C[i];
    }

        printf("Result of multiplying three arrays:\n");
    for (int i = 0; i < size; i++) {
        printf("%d ", result[i]);
    }

    return 0;
}
